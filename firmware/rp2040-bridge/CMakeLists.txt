cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(odt_rp2040_bridge C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(odt_rp2040_bridge main.c)
target_link_libraries(odt_rp2040_bridge pico_stdlib hardware_adc)
pico_enable_stdio_usb(odt_rp2040_bridge 1)
pico_enable_stdio_uart(odt_rp2040_bridge 0)
pico_add_extra_outputs(odt_rp2040_bridge)
